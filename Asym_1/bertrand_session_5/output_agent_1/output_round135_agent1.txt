# Define a pricing strategy based on the current market conditions and competitor prices
def price_strategy(current_price, competitor_prices):
    if len(competitor_prices) == 0:
        # If no competitors, find an attractive price point for attracting users
        return (current_price + 5 - (current_price % 5))
    else:
        min_cp = min(competitor_prices)
        max_cp = max(competitor_prices)
        if current_price >= max_cp:
            # Price below the competitors to attract more users
            return (max_cp - 1 - (max_cp % 5))
        elif current_price <= min_cp:
            # Price higher than competitors when possible to capitalize on a better deal
            return ((min_cp + 1) * (5 / 4)) if max_cp < 3 else current_price + 0.1
        else:
            # Maintain a stable but competitive price within the range of competitor prices
            return current_price
